# Top-level EDK2 (UEFI/OVMF) syzlang description.
#
# The fuzzer never emits raw UEFI calls; instead it emits a single
# pseudo-syscall, syz_edk2_run_program, whose argument is an array of
# tagged commands (syz_edk2_call) that the in-firmware SyzAgentDxe driver
# interprets and dispatches to the corresponding Boot Services / Runtime
# Services / protocol entry points.
#
# IDs in the syz_edk2_call union MUST match the SYZ_EDK2_API_* enum in
# OvmfPkg/SyzAgentDxe/SyzAgentDxe.h on the EDK2 side. The same convention
# is used by SYZOS; see docs/syzos.md and docs/edk2_design.md.
#
# Every payload struct is marked [packed] so the in-memory layout the
# prog package builds matches the C structs SyzAgentDxe casts to. The
# tools/syz-edk2-fuzz host translator walks the *prog.Prog tree directly
# and emits raw bytes; alignment is therefore syzlang's responsibility.

# EDK2 headers are not extractable by syz-extract today; the constants below
# are provided directly in edk2.txt.const. See docs/edk2_design.md §4.1.5.
meta noextract
meta arches["amd64"]

# Memory mapping for the host executor's data segment. The implementation
# in executor/common.h forwards to mmap() because the edk2 executor runs on
# the Linux host (HostFuzzer mode); see executor_edk2.h.
syz_mmap(addr vma, len len[addr])

syz_edk2_run_program(prog ptr[in, edk2_program])

edk2_program {
	magic	const[EDK2_PROGRAM_MAGIC, int32]
	ncalls	len[calls, int32]
	calls	array[syz_edk2_call, 1:32]
} [packed]

# Wrapper applied to every command. NUM is the SYZ_EDK2_API_* enum value;
# size is the byte length of (call+size+payload), exactly like SYZOS's
# syzos_api[] template.
type syz_edk2_api[NUM, PAYLOAD] {
	call	const[NUM, int32]
	size	bytesize[parent, int32]
	payload	PAYLOAD
} [packed]

# A no-op command, useful as a baseline for the dispatch loop and for
# tests. Decorated as a single int64 value so the agent can echo it back.
edk2_api_nop_payload {
	cookie	int64
} [packed]

# ----------------------------------------------------------------------
# Variable Services (gRT->{Get,Set,QueryVariableInfo}Variable + friends).
# ----------------------------------------------------------------------
edk2_api_set_variable {
	name_size	bytesize[name, int16]
	attributes	flags[edk2_variable_attrs, int32]
	data_size	bytesize[data, int16]
	namespace	flags[edk2_variable_namespace, int32]
	name		array[int16, 1:32]
	data		array[int8, 0:256]
} [packed]

edk2_api_get_variable {
	name_size	bytesize[name, int16]
	max_data	int16[0:512]
	namespace	flags[edk2_variable_namespace, int32]
	name		array[int16, 1:32]
} [packed]

edk2_api_query_variable_info {
	attributes	flags[edk2_variable_attrs, int32]
} [packed]

# Iterator-style enumeration. The agent maintains a small bounded
# Name buffer and feeds it back into GetNextVariableName until the
# fuzzer asks it to stop. Heavy on the variable-store name parser.
edk2_api_get_next_variable_name {
	max_name	int16[8:256]
	reset		int8[0:1]
	pad0		const[0, int8]
	pad1		const[0, int8]
	pad2		const[0, int8]
} [packed]

edk2_variable_attrs = EFI_VARIABLE_NON_VOLATILE, EFI_VARIABLE_BOOTSERVICE_ACCESS, EFI_VARIABLE_RUNTIME_ACCESS, EFI_VARIABLE_HARDWARE_ERROR_RECORD, EFI_VARIABLE_AUTHENTICATED_WRITE_ACCESS, EFI_VARIABLE_TIME_BASED_AUTHENTICATED_WRITE_ACCESS, EFI_VARIABLE_APPEND_WRITE

# Symbolic names for well-known variable namespaces. The agent maps each
# id back to the appropriate EFI_GUID via a lookup table baked into
# SyzAgentDxe. EDK2_VAR_NS_SYZ is the default scratch namespace.
edk2_variable_namespace = EDK2_VAR_NS_SYZ, EDK2_VAR_NS_GLOBAL, EDK2_VAR_NS_IMAGE_SECURITY_DB, EDK2_VAR_NS_IMAGE_SECURITY_DBX, EDK2_VAR_NS_IMAGE_SECURITY_DBT

# ----------------------------------------------------------------------
# Memory Services (gBS->{AllocatePool,FreePool,AllocatePages,FreePages,
# CopyMem,SetMem}).
# ----------------------------------------------------------------------
edk2_api_allocate_pool {
	mem_type	flags[edk2_memory_type, int32]
	size		int32[1:65536]
} [packed]

edk2_api_free_pool {
# Index into the agent's allocation table; the agent maps it to a
# real EFI pointer. We do not let the fuzzer hand the firmware raw
# pointers (see docs/edk2_design.md §4.2.1).
	alloc_index	int32[0:31]
} [packed]

edk2_api_allocate_pages {
	alloc_type	flags[edk2_allocate_type, int32]
	mem_type	flags[edk2_memory_type, int32]
	pages		int32[1:64]
} [packed]

edk2_api_free_pages {
	alloc_index	int32[0:31]
} [packed]

# CopyMem / SetMem operate inside a previously-allocated agent slot,
# bounded by (offset, length). The agent clamps both at the slot extent.
edk2_api_copy_mem {
	dst_index	int32[0:31]
	src_index	int32[0:31]
	dst_offset	int32[0:65535]
	src_offset	int32[0:65535]
	length		int32[0:65536]
} [packed]

edk2_api_set_mem {
	alloc_index	int32[0:31]
	offset		int32[0:65535]
	length		int32[0:65536]
	value		int8
	pad0		const[0, int8]
	pad1		const[0, int8]
	pad2		const[0, int8]
} [packed]

# CalculateCrc32 over a previously-allocated buffer. Cheap, deterministic,
# stresses the table-driven CRC routine.
edk2_api_calculate_crc32 {
	alloc_index	int32[0:31]
	offset		int32[0:65535]
	length		int32[0:65536]
} [packed]

edk2_memory_type = EfiReservedMemoryType, EfiLoaderCode, EfiLoaderData, EfiBootServicesCode, EfiBootServicesData, EfiRuntimeServicesCode, EfiRuntimeServicesData, EfiConventionalMemory, EfiACPIReclaimMemory, EfiACPIMemoryNVS, EfiPalCode, EfiPersistentMemory

edk2_allocate_type = AllocateAnyPages, AllocateMaxAddress, AllocateAddress

# SetTime takes a fully-formed EFI_TIME on the wire (16 bytes packed).

	month	int8[1:12]
	day	int8[1:31]
	hour	int8[0:23]
	minute	int8[0:59]
	second	int8[0:59]
	pad0	const[0, int8]
	nano	int32[0:999999999]
	tz	int16
	dl	int8
	pad1	const[0, int8]
} [packed]

edk2_api_stall {
	microseconds	int32[0:5000]

edk2_api_set_watchdog_timer {

# ----------------------------------------------------------------------
# HII Database (gBS->LocateProtocol(HiiDatabase) + NewPackageList /

# ----------------------------------------------------------------------

	package_size	bytesize[data, int16]
	data		array[int8, 4:512]
} [packed]

edk2_api_hii_remove_package_list {
	handle_index	int32[0:31]
} [packed]

# ----------------------------------------------------------------------
# AddressSanitizer integration. These commands ride on top of the EDK2-side
# AsanSyz facade exposed by SyzAgentDxe when the firmware is built with both
# SYZ_AGENT_ENABLE=TRUE and ASAN_ENABLE=TRUE. They poke at shadow memory
# inside a previously-allocated AllocatePool / AllocatePages slot only;
# raw pointers are never sent across the wire.
# ----------------------------------------------------------------------
edk2_api_asan_payload {
	alloc_index	int32[0:31]
	offset		int32[0:65535]
	length		int32[1:65536]
	is_write	int8[0:1]
	pad0		const[0, int8]
	pad1		const[0, int8]
	pad2		const[0, int8]
} [packed]
